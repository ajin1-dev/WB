## Prevalence of undernourishment (% of population); ind = SN.ITK.DEFC.ZS

# Load in FIES Dataset
## Prevalence of moderate or severe food insecurity in the population (%); ind = SN.ITK.MSFI.ZS
insecurity_ms<- read.csv("Somalia_Hunger/Data/API_SN.ITK.MSFI.ZS_DS2_en_csv_v2_7955.csv", skip = 4) |>
  select(c(Country.Name, Country.Code, "X2021")) |>
  rename(pct = X2021)

## Creating Makeshift Benchmarks
### world average
wld_avg_ms <- country_pops |>
  filter(year == "2021") |>
  right_join(insecurity_ms, by = c("code" = "Country.Code")) |>
  summarize(pct = weighted.mean(pct, population, na.rm = TRUE)) |>
  mutate(Country.Name = "World", code = "WLD")
### ssa average
ssa_avg_ms <- country_pops |>
  filter(year == "2021") |>
  right_join(insecurity_ms, by = c("code" = "Country.Code")) |>
  filter(code %in% ssa_codes) |>
  summarize(pct = weighted.mean(pct, population, na.rm = TRUE)) |>
  mutate(Country.Name = "Sub-Saharan Africa", Country.Code = "SSF")
### low income average
lic_avg_ms <- country_pops |>
  filter(year == "2021") |>
  right_join(insecurity_ms, by = c("code" = "Country.Code")) |>
  filter(code %in% low_inc_codes) |>
  summarize(pct = weighted.mean(pct, population, na.rm = TRUE)) |>
  mutate(Country.Name = "Low income", Country.Code = "LIC")
### fcs average
fcs_avg_ms <- country_pops |>
  filter(year == "2021") |>
  right_join(insecurity_ms, by = c("code" = "Country.Code")) |>
  filter(code %in% fcs_codes) |>
  summarize(pct = weighted.mean(pct, population, na.rm = TRUE)) |>
  mutate(Country.Name = "FCS Countries", Country.Code = "FCS")
### Joining to main dataset and filtering
insecurity_ms <- insecurity_ms |>
  filter(Country.Code == "SOM") |>
  bind_rows( wld_avg_ms, ssa_avg_ms, lic_avg_ms, fcs_avg_ms)

ipc0 <- read_excel("Somalia_Hunger/Data/IPC_Population_Analysis_Styled_2017,2027_1781118839874.xlsx") |>

  filter(!is.na(`Country Population`) & !is.na(`Current - Analysis Period`)) |>

ipc1 <- read_excel("Somalia_Hunger/Data/IPC_Population_Analysis_Styled_2017,2027_1781119874050.xlsx") |>

  filter(!is.na(`Country Population`))

  rename(population = `Country Population`, date = `Date of Analysis`) |>
  mutate(date = ymd_hms(date))


## Data with population
ipc_pop <- ipc |>
  select(population, date, ends_with("Pop")) |>
  rename_with(~gsub("Current - ", "", .x)) |>
  rename_with(~gsub(" Pop", "", .x))

data <- wbstats::wb_data(indicator = "SN.ITK.DEFC.ZS", lang = "en", country = "countries_only")
progress_results <- track_progress(data = data,
                                   indicator = "SN.ITK.DEFC.ZS",
                                   code_col = "iso3c",
                                   year_col = "date",
                                   startyear_data = 2001,
                                   endyear_data = 2023,
                                   eval_from = 2018,
                                   eval_to = 2023, 
                                   speed = TRUE,
                                   percentiles = FALSE,
                                   future = FALSE, ## TRUE,
                                   target_year = 2030,
                                   sequence_pctl = seq(20,80,20),
                                   sequence_speed = c(0.25, 0.5, 1, 2, 4),
                                   best = "low",
                                   support = 1,
                                   granularity = 0.01)

path_speed  <- progress_results$predicted_changes$path_speed
changes_speed <- progress_results$predicted_changes$changes_speed

typical_value <- progress_results$path_historical$speed
data_model <- progress_results$data_model$data_model
historical_path_speed <- progress_results$path_historical$speed
scores <- progress_results$scores$speed

future <- progress_results$path_future$speed

score_speed <- classifications |>
  select(Economy, Code) |>
  right_join(progress_results$scores$speed, by = c("Code" = "code")) |>
  rename("country" = Economy, "code" = Code)

score_speed_fcs_separate <- score_speed |>
  filter(code %in% fcs_codes) |>
  mutate(country = fct_reorder(country, score)) |>
  mutate(issomalia = (code == "SOM"))

fcs_score_plot <- score_speed_fcs_separate |>
  ggplot(aes(x = country, y = score, fill = issomalia)) + geom_bar(stat="identity") +
  coord_flip() +
  labs(
    title = "Rate of Progress from 2018-2023",
    subtitle = "2026 FCS Countries",
    y = "Rate of Progress",
    x = ""
  ) +
  theme(legend.position = "none")
fcs_score_plot



score_speed_fcs_unweighted <- score_speed |>
  filter(code %in% fcs_codes) |>
  summarize(score = mean(score, na.rm = TRUE)) |>
  mutate(Economy = "FCS Countries", Country.Code = "FCS")






score_speed_fcs_weighted <- score_speed |>
  filter(code %in% fcs_codes) |>
  left_join(country_pops, by = "code") |>
  mutate(score = score*population) |>
  summarize(score = sum(score, na.rm = TRUE),
            population = sum(population, na.rm = FALSE)) |>
  mutate(score = score/population)
  mutate(Economy = "FCS Countries", Country.Code = "FCS")
